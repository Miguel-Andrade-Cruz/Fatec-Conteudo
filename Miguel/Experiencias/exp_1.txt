#ifdef exc6
// Pinos de ativação

int led_ext = 12;
int led_int = LED_BUILTIN;

int tick_size = 500;
int current_time;
int current_tick;

// time in ticks
int int_cycle_on = 5;
int int_cycle_off = 9;
int int_full_cycle = int_cycle_on + int_cycle_off;

int ext_cycle_on = 6;
int ext_cycle_off = 10;
int ext_full_cycle = ext_cycle_on + ext_cycle_off;


void setup() {
  
  pinMode(led_int, OUTPUT);
  pinMode(led_ext, OUTPUT);
  
}


void loop() {
  
  current_time = millis();
  current_tick = current_time / tick_size;
  
  if (current_tick % int_full_cycle < int_cycle_on) {
    digitalWrite(led_int, HIGH);
  } else {
    digitalWrite(led_int, LOW);
  }
  
  if (current_tick % ext_full_cycle < ext_cycle_on) {
    digitalWrite(led_ext, HIGH);
  } else {
    digitalWrite(led_ext, LOW);
  }
}
#endif
